// Zero-copy token scanning across multi-megabyte Markdown AST nodes
pub fn scan_markdown_ast<'a>(&mut self, raw_stream: &'a [u8]) -> PyResult<usize> {
let simd_vectors = raw_stream.chunks_exact(64);
let mut token_count = 0usize;
for chunk in simd_vectors {
let lanes = u8x64::from_slice(chunk);
let mask = lanes.simd_eq(u8x64::splat(b'\n'));
token_count += mask.to_bitmask().count_ones() as usize;
}
Ok(token_count)
}
}
